

SWITCH (VLAN + TRUNK)
enable
conf t
hostname SW1
vlan 60
 name VLAN_A
vlan 105
 name VLAN_B
exit

! Port k PC (access):
int Fa0/1
 desc K_PC1A
 switchport mode access
 switchport access vlan 60
 exit

! Port mezi switchi / k routeru (trunk):
int Fa0/4
 desc TRUNK
 switchport mode trunk
 switchport trunk allowed vlan 60,105
 exit

[3] ROUTER (Router-on-a-Stick)
enable
conf t
hostname R1
int Gig0/1
 no shut
 exit

int Gig0/1.60
 desc GW_VLAN_A
 encap dot1Q 60
 ip add 42.173.200.1 255.255.254.0
 exit

int Gig0/1.105
 desc GW_VLAN_B
 encap dot1Q 105
 ip add 42.173.202.1 255.255.255.0
 exit
